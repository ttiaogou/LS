cmake_minimum_required(VERSION 3.22)
project(location_sensor_probe LANGUAGES C CXX ASM)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(uid_pattern STATIC uid_pattern.cpp uid_receiver.cpp arm64_flow.cpp
    elf_function.cpp elf_regions.cpp eh_frame.cpp code_closure.cpp mapped_file.cpp passthrough.cpp
    step_stream.cpp step_wire.cpp step_delivery.cpp)
target_include_directories(uid_pattern PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(uid_pattern PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_options(uid_pattern PRIVATE -Wall -Wextra -Werror)

if(ANDROID)
    if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
        message(FATAL_ERROR "The diagnostic locator currently supports arm64-v8a only")
    endif()
    # SENSOR_PROBE_JNI_PACKAGE 是 Java 类所在包的下划线形式，由 Gradle 传入
    # （见 sensor-native/build.gradle.kts）。这里刻意不设为必填：纯核心测试与独立诊断构建
    # 不编译 JNI 桥，配置阶段不该被挡；真正编译 jni_bridge.cpp 时缺定义会由源码里的 #error
    # 明确报错，独立构建按 README 的命令显式传该定义即可。
    add_library(location_accel SHARED jni_bridge.cpp target_locator.cpp)
    if(DEFINED SENSOR_PROBE_JNI_PACKAGE AND NOT SENSOR_PROBE_JNI_PACKAGE STREQUAL "")
        target_compile_definitions(location_accel PRIVATE SENSOR_PROBE_JNI_PACKAGE=${SENSOR_PROBE_JNI_PACKAGE})
    endif()
    target_link_libraries(location_accel PRIVATE uid_pattern log dl)
    target_compile_options(location_accel PRIVATE -Wall -Wextra -Werror -fno-exceptions)
    target_link_options(location_accel PRIVATE -Wl,--no-undefined)
    option(SENSOR_PROBE_PASSTHROUGH "Install a diagnostic passthrough hook (no simulation)" OFF)
    option(SENSOR_PROBE_STEPS "Experimental non-wake step delivery (requires passthrough)" OFF)
    option(SENSOR_PROBE_EVENT_DUMP "Log the first real step events for layout verification" OFF)
    if(SENSOR_PROBE_STEPS AND NOT SENSOR_PROBE_PASSTHROUGH)
        message(FATAL_ERROR "Step delivery requires the forwarding hook")
    endif()
    if(SENSOR_PROBE_PASSTHROUGH)
        include(${CMAKE_CURRENT_SOURCE_DIR}/shadowhook_source.cmake)
        target_sources(location_accel PRIVATE passthrough_hook.cpp)
        target_link_libraries(location_accel PRIVATE shadowhook)
        target_compile_definitions(location_accel PRIVATE SENSOR_PROBE_PASSTHROUGH=1)
        if(SENSOR_PROBE_EVENT_DUMP)
            target_compile_definitions(location_accel PRIVATE SENSOR_PROBE_EVENT_DUMP=1)
        endif()
        if(SENSOR_PROBE_STEPS)
            target_sources(location_accel PRIVATE step_hook.cpp)
            target_compile_definitions(location_accel PRIVATE SENSOR_PROBE_STEPS=1)
        endif()
    endif()
    option(SENSOR_PROBE_DEVICE_TESTS "Build standalone ARM64 core tests (no hook installation)" OFF)
    if(SENSOR_PROBE_DEVICE_TESTS)
        foreach(suite uid_pattern uid_receiver eh_frame code_closure elf_function passthrough step_stream step_wire step_delivery)
            add_executable(${suite}_test tests/${suite}_test.cpp)
            target_link_libraries(${suite}_test PRIVATE uid_pattern)
            target_compile_options(${suite}_test PRIVATE -Wall -Wextra -Werror)
        endforeach()
        add_executable(sensor_probe_file probe_file.cpp)
        target_link_libraries(sensor_probe_file PRIVATE uid_pattern)
        target_compile_options(sensor_probe_file PRIVATE -Wall -Wextra -Werror)
    endif()
else()
    enable_testing()
    option(SENSOR_PROBE_SANITIZERS "Enable host address/undefined behavior checks" OFF)
    if(SENSOR_PROBE_SANITIZERS)
        target_compile_options(uid_pattern PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(uid_pattern PUBLIC -fsanitize=address,undefined)
    endif()
    add_executable(uid_pattern_test tests/uid_pattern_test.cpp)
    target_link_libraries(uid_pattern_test PRIVATE uid_pattern)
    target_compile_options(uid_pattern_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME uid_pattern COMMAND uid_pattern_test)
    add_executable(uid_receiver_test tests/uid_receiver_test.cpp)
    target_link_libraries(uid_receiver_test PRIVATE uid_pattern)
    target_compile_options(uid_receiver_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME uid_receiver COMMAND uid_receiver_test)
    foreach(suite eh_frame code_closure step_stream step_wire step_delivery)
        add_executable(${suite}_test tests/${suite}_test.cpp)
        target_link_libraries(${suite}_test PRIVATE uid_pattern)
        target_compile_options(${suite}_test PRIVATE -Wall -Wextra -Werror)
        add_test(NAME ${suite} COMMAND ${suite}_test)
    endforeach()
    add_executable(elf_function_test tests/elf_function_test.cpp)
    target_link_libraries(elf_function_test PRIVATE uid_pattern)
    target_compile_options(elf_function_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME elf_function COMMAND elf_function_test)
    add_executable(passthrough_test tests/passthrough_test.cpp)
    target_link_libraries(passthrough_test PRIVATE uid_pattern)
    target_compile_options(passthrough_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME passthrough COMMAND passthrough_test)
    add_executable(event_dump_test tests/event_dump_test.cpp passthrough_hook.cpp)
    target_include_directories(event_dump_test PRIVATE tests/shadowhook-stub tests/android-log-stub)
    target_compile_definitions(event_dump_test PRIVATE SENSOR_PROBE_EVENT_DUMP=1)
    target_link_libraries(event_dump_test PRIVATE uid_pattern)
    target_compile_options(event_dump_test PRIVATE -Wall -Wextra -Werror)
    add_test(NAME event_dump COMMAND event_dump_test)
    add_executable(passthrough_hook_test tests/passthrough_hook_test.cpp passthrough_hook.cpp)
    target_include_directories(passthrough_hook_test PRIVATE tests/shadowhook-stub)
    target_link_libraries(passthrough_hook_test PRIVATE uid_pattern)
    target_compile_options(passthrough_hook_test PRIVATE -Wall -Wextra -Werror)
    foreach(scenario success init-failure hook-failure missing-trampoline)
        add_test(NAME hook-${scenario} COMMAND passthrough_hook_test ${scenario})
    endforeach()
    add_executable(step_hook_test tests/step_hook_test.cpp step_hook.cpp)
    target_include_directories(step_hook_test PRIVATE tests/shadowhook-stub)
    target_link_libraries(step_hook_test PRIVATE uid_pattern)
    target_compile_options(step_hook_test PRIVATE -Wall -Wextra -Werror)
    foreach(scenario success both-variants missing-symbol remove-failure destroy-failure missing-trampoline)
        add_test(NAME steps-${scenario} COMMAND step_hook_test ${scenario})
    endforeach()
    add_executable(sensor_probe_file probe_file.cpp)
    target_link_libraries(sensor_probe_file PRIVATE uid_pattern)
    target_compile_options(sensor_probe_file PRIVATE -Wall -Wextra -Werror)
endif()
